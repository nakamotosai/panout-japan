+      axisLine: 'rgba(255,255,255,0.16)', axisLabel: '#8B8D94',
+      splitLine: 'rgba(255,255,255,0.08)', splitLineType: 'dashed',
+      zoomBorder: 'rgba(255,255,255,0.14)', zoomBg: '#0E0F14', zoomFill: 'rgba(232,232,236,0.12)', zoomHandle: '#E8E8EC', zoomText: '#8B8D94',
+      markBorder: 'rgba(14,15,20,0.92)', pointBorder: '#06070A'
+    }
+  }
+};
+
+let chart = null;
+let rawData = null;
+let activeRange = '10y';   // 默认近10年
+let focusCabinet = null;   // 点击图例聚焦的首相 id，null=全部
+
+/* 主题下拉 */
+const themeMenu = document.getElementById('themeMenu');
+function renderThemeMenu() {
+  themeMenu.innerHTML = '';
+  THEMES.forEach(t => {
+    const b = document.createElement('button');
+    b.type = 'button';
+    b.dataset.theme = t.id;
+    b.innerHTML = '<span class="sw" style="background:' + t.swatch + '"></span>' + t.label;
+    b.classList.toggle('active', t.id === currentTheme);
+    b.addEventListener('click', () => {
+      applyTheme(t.id);
+      renderChart();
+      closeThemeMenu();
+    });
+    themeMenu.appendChild(b);
+  });
+}
+function openThemeMenu() { themeMenu.classList.add('open'); }
+function closeThemeMenu() { themeMenu.classList.remove('open'); }
+document.getElementById('themeBtn').addEventListener('click', (e) => {
+  e.stopPropagation();
+  themeMenu.classList.toggle('open');
+});
+document.addEventListener('click', () => closeThemeMenu());
+
+let currentTheme = localStorage.getItem('cabinet-theme') || 'gruvbox-dark'; // 默认 gruvbox
+function applyTheme(id) {
+  currentTheme = id;
+  document.documentElement.dataset.theme = id;
+  localStorage.setItem('cabinet-theme', id);
+  const meta = THEMES.find(t => t.id === id);
+  document.getElementById('themeBtnLabel').textContent = 'テーマ：' + meta.label;
+  document.getElementById('themeSwatch').style.background = meta.swatch;
+}
+
+/* 时间分段按钮 */
+const RANGES = [
+  { id: '3m',  label: '3ヶ月' },
+  { id: '1y',  label: '1年' },
+  { id: '2y',  label: '2年' },
+  { id: '5y',  label: '5年' },
+  { id: '10y', label: '10年' },
+  { id: 'all', label: '全期間' }
+];
+const RANGE_WINDOW_MONTHS = { '3m': 3, '1y': 12, '2y': 24, '5y': 60, '10y': 120, 'all': 0 };
+const zoomSeg = document.getElementById('zoomSeg');
+function renderZoomSeg() {
+  zoomSeg.innerHTML = '';
+  RANGES.forEach(r => {
+    const b = document.createElement('button');
+    b.type = 'button';
+    b.textContent = r.label;
+    b.classList.toggle('active', r.id === activeRange);
+    b.addEventListener('click', () => {
+      activeRange = r.id;
+      focusCabinet = null;   // 手动切时间窗=退出首相聚焦
+      renderZoomSeg();
+      buildLegend();
+      renderChart();
+    });
+    zoomSeg.appendChild(b);
+  });
+}

+function tchart() { return THEME_CHART[currentTheme].chart; }
+function cabColors() { return THEME_CHART[currentTheme].palette; }

+async function loadData() {
+  try {
+    const resp = await fetch('data.json');
+    rawData = await resp.json();
+  } catch (e) {
+    console.error('data.json load failed', e);
+    return;
+  }
+  renderThemeMenu();
+  renderZoomSeg();
+  applyTheme(currentTheme);
+  buildLegend();
+  renderChart();
+}

+function buildLegend() {
+  const el = document.getElementById('legend');
+  el.innerHTML = '';
+
+  // 退出聚焦按钮：仅在聚焦某位首相时显示
+  if (focusCabinet) {
+    const exitBtn = document.createElement('button');
+    exitBtn.type = 'button';
+    exitBtn.className = 'pill exit-focus';
+    exitBtn.textContent = '✕ 全部表示';
+    exitBtn.title = 'クリックで全首相の線を元に戻す';
+    exitBtn.addEventListener('click', () => {
+      focusCabinet = null;
+      buildLegend();
+      renderChart();
+    });
+    el.appendChild(exitBtn);
+  }
+
+  rawData.cabinets.forEach((c, i) => {
+    const pill = document.createElement('button');
+    pill.type = 'button';
+    pill.className = 'pill';
+    pill.title = 'クリックでこの首相の線にフォーカス';
+    pill.textContent = c.pm;
+    const color = cabColors()[i % cabColors().length];
+    pill.style.borderLeft = '3px solid ' + color;
+    pill.style.color = color;
+    if (focusCabinet === c.id) pill.classList.add('focused');
+    pill.addEventListener('click', () => {
+      focusCabinet = (focusCabinet === c.id) ? null : c.id;
+      buildLegend();
+      renderChart();
+    });
+    el.appendChild(pill);
+  });
+}

+function pmByDate(dateStr) {
+  const d = new Date(dateStr);
+  for (let i = rawData.cabinets.length - 1; i >= 0; i--) {
+    if (d >= new Date(rawData.cabinets[i].start)) return rawData.cabinets[i];
+  }
+  return rawData.cabinets[0];
+}
+
+function renderChart() {
+  if (!chart) chart = echarts.init(document.getElementById('chart'), null, {renderer: 'canvas'});
+  if (!rawData) return;
+
+  const polls = rawData.polls;
+  const cabinets = rawData.cabinets;
+  const C = cabColors();
+  const T = tchart();
+
+  // (内閣, 年月) バケットで実測点を実調査日で所属判別 → 算術平均
+  const buckets = {};
+  polls.forEach(p => {
+    if (p.derived) return;
+    const cab = pmByDate(p.date);
+    const ym = p.date.slice(0, 7);
+    const key = cab.id + '|' + ym;
+    if (!buckets[key]) buckets[key] = { cab, ym, sums: 0, n: 0, sources: [] };
+    buckets[key].sums += p.value;
+    buckets[key].n += 1;
+    buckets[key].sources.push({ src: p.source, v: p.value, kind: p.kind || '' });
+  });
+
+  const series = [];
+  cabinets.forEach((c, i) => {
+    const color = C[i % C.length];
+    const points = [];
+    Object.keys(buckets).sort().forEach(key => {
+      const rec = buckets[key];
+      if (rec.cab.id !== c.id) return;
+      const d = rec.ym + '-15';
+      const avg = Math.round((rec.sums / rec.n) * 10) / 10;
+      points.push([d, avg, rec.n, rec.sources]);
+    });
+
+    if (points.length > 0) {
+      const focused = focusCabinet === c.id;
+      series.push({
+        name: c.pm,
+        type: 'line',
+        data: points,
+        showSymbol: true,
+        symbol: 'circle',
+        symbolSize: focused ? 7 : 5,
+        smooth: false,
+        connectNulls: false,
+        lineStyle: { color: color, width: focused ? 4 : 2, type: 'solid', opacity: focusCabinet ? (focused ? 1 : 0.22) : 1 },
+        itemStyle: { color: color, borderColor: T.pointBorder, borderWidth: 1, opacity: focusCabinet ? (focused ? 1 : 0.22) : 1 },
+        emphasis: { focus: 'series', scale: 1.4 },
+        z: focused ? 5 : 2,
+        markLine: {
+          silent: true,
+          symbol: ['none','none'],
+          lineStyle: { color: color, type: 'solid', width: focused ? 2 : 1.2, opacity: focusCabinet ? (focused ? 0.9 : 0.15) : 0.55 },
+          label: {
+            position: 'end',
+            formatter: c.pm,
+            color: color, fontSize: 14, fontWeight: 700,
+            padding: [3,8],
+            backgroundColor: T.markBorder,
+            borderColor: color, borderWidth: 1, borderRadius: 4,
+            distance: 6
+          },
+          data: [{ xAxis: c.start }]
+        }
+      });
+    }
+  });
+
+  const allDates = polls.map(p => p.date).sort();
+  const startDate = allDates[0];
+  const endDate = allDates[allDates.length - 1];
+
+  // 近N年窗口（动态）：按实际时间跨度比例换算 dataZoom start
+  const wMonths = RANGE_WINDOW_MONTHS[activeRange];
+  let zoomStart = 0;
+  if (wMonths > 0) {
+    const spanMs = new Date(endDate) - new Date(startDate);
+    const winMs = wMonths * 30.44 * 86400000;
+    zoomStart = Math.max(0, Math.round((1 - winMs / spanMs) * 1000) / 10);
+  }
+
+  // 聚焦首相：窗口平移到其任期（中心对齐），但宽度保持 activeRange 窗口不变（10年就是10年）
+  const focusCab = focusCabinet ? rawData.cabinets.find(c => c.id === focusCabinet) : null;
+  let focusWindow = null;   // { startValue, endValue } 或 null
+  if (focusCab && wMonths > 0) {
+    const winMs = wMonths * 30.44 * 86400000;
+    const tMin = new Date(startDate).getTime();
+    const tMax = new Date(endDate).getTime();
+    const cStart = new Date(focusCab.start).getTime();
+    const cEnd = new Date(focusCab.end).getTime();
+    const center = (cStart + cEnd) / 2;
+    let ws = center - winMs / 2;
+    let we = center + winMs / 2;
+    // clamp 到数据范围，窗口宽度恒等于 winMs（数据跨度26.5年 > 10年，恒有解）
+    if (ws < tMin) { ws = tMin; we = tMin + winMs; }
+    if (we > tMax) { we = tMax; ws = tMax - winMs; }
+    focusWindow = { startValue: new Date(ws), endValue: new Date(we) };
+  }
+
+  const sliderDZ = {
+    type: 'slider', xAxisIndex: 0,
+    left: 380, right: 12,
+    height: 24, bottom: 18,
+    borderRadius: 8,
+    borderColor: T.zoomBorder,
+    backgroundColor: T.zoomBg,
+    fillerColor: T.zoomFill,
+    handleSize: 14,
+    brushSelect: false,
+    dataBackground: {
+      lineStyle: { color: 'rgba(0,0,0,0)' },
+      areaStyle: { color: 'rgba(0,0,0,0)' }
+    },
+    selectedDataBackground: {
+      lineStyle: { color: 'rgba(0,0,0,0)' },
+      areaStyle: { color: 'rgba(0,0,0,0)' }
+    },
+    moveHandleSize: 0,
+    // 实心圆角滑块（替换 ECharts 默认梯形手柄）+ 描边阴影小装饰
+    handleIcon: 'path://M-8,-9h16a4,4 0 0 1 4,4v10a4,4 0 0 1 -4,4h-16a4,4 0 0 1 -4,-4v-10a4,4 0 0 1 4,-4z',
+    handleStyle: {
+      color: T.zoomHandle,
+      borderColor: 'rgba(0,0,0,0.28)',
+      borderWidth: 1,
+      shadowBlur: 5,
+      shadowColor: 'rgba(0,0,0,0.35)',
+      shadowOffsetY: 1.5
+    },
+    emphasis: { handleStyle: { color: T.zoomHandle, borderColor: '#fff', shadowBlur: 7 } },
+    textStyle: { color: T.zoomText, fontSize: 10 }
+  };
+  if (focusWindow) {
+    sliderDZ.startValue = focusWindow.startValue;
+    sliderDZ.endValue = focusWindow.endValue;
+  } else {
+    sliderDZ.start = zoomStart;
+    sliderDZ.end = 100;
+  }
+
+  const option = {
+    backgroundColor: 'transparent',
+    grid: { left: 64, right: 32, top: 30, bottom: 100, borderColor: T.gridBorder },
+    legend: { show: false },
+    tooltip: {
+      trigger: 'axis',
+      backgroundColor: T.tooltipBg,
+      borderColor: T.tooltipBorder,
+      textStyle: { color: T.tooltipText, fontSize: 12 },
+      formatter: function(params) {
+        if (!params || params.length === 0) return '';
+        const d = new Date(params[0].axisValue);
+        const ym = d.getFullYear() + '年' + (d.getMonth()+1) + '月';
+        const p = params[0];
+        const avg = p.data ? p.data[1] : null;
+        const n = p.data ? p.data[2] : 0;
+        const sources = p.data ? (p.data[3] || []) : [];
+        let html = '<div style="font-weight:700;margin-bottom:4px;font-size:13px">' + ym + ' 平均 <b>' + avg + '%</b></div>';
+        html += '<div style="color:' + T.tooltipMuted + ';font-size:11px;margin-bottom:6px">' + n + '社平均</div>';
+        sources.forEach(s => {
+          html += '<span style="color:' + T.tooltipDim + '">▪</span> ' + s.src + ': <b>' + s.v + '%</b>'
+                + (s.kind ? ' <span style="color:' + T.tooltipDim + ';font-size:10px">(' + s.kind + ')</span>' : '') + '<br>';
+        });
+        return html;
+      }
+    },
+    xAxis: {
+      type: 'time', min: startDate, max: endDate,
+      axisLine: { lineStyle: { color: T.axisLine } },
+      axisLabel: { color: T.axisLabel, fontSize: 11, hideOverlap: true },
+      splitLine: { show: true, lineStyle: { color: T.splitLine, type: T.splitLineType } }
+    },
+    yAxis: {
+      type: 'value', min: 0, max: 100, interval: 10,
+      name: '支持率 (%)',
+      nameTextStyle: { color: T.axisLabel, fontSize: 11 },
+      axisLine: { show: false },
+      axisLabel: { color: T.axisLabel, formatter: '{value}%' },
+      splitLine: { show: true, lineStyle: { color: T.splitLine, type: T.splitLineType } }
+    },
+    dataZoom: [
+      sliderDZ,
+      {
+        type: 'inside', xAxisIndex: 0,
+        start: focusWindow ? undefined : zoomStart,
+        end: focusWindow ? undefined : 100,
+        startValue: focusWindow ? focusWindow.startValue : undefined,
+        endValue: focusWindow ? focusWindow.endValue : undefined,
+        zoomOnMouseWheel: 'shift', moveOnMouseWheel: true
+      }
+    ],
+    animation: false,
+    series: series
+  };
+
+  chart.setOption(option, true);
+}
+
+/* 点击图表空白关闭菜单兜底 */
+document.addEventListener('keydown', (e) => { if (e.key === 'Escape') closeThemeMenu(); });
+
+window.addEventListener('resize', () => { if (chart) chart.resize(); });
+loadData();
+</script>
